116static uint8_t measure_distance_cm(uint32_t *distance_cm)
117{
118 uint16_t pulse_start;
119 uint16_t pulse_width;
120
121 HAL_GPIO_WritePin(APP_HCSR04_TRIGGER_PORT, APP_HCSR04_TRIGGER_PIN,
122 GPIO_PIN_RESET);
123 delay_us(2);
124 HAL_GPIO_WritePin(APP_HCSR04_TRIGGER_PORT, APP_HCSR04_TRIGGER_PIN,
125 GPIO_PIN_SET);
126 delay_us(10);
127 HAL_GPIO_WritePin(APP_HCSR04_TRIGGER_PORT, APP_HCSR04_TRIGGER_PIN,
128 GPIO_PIN_RESET);
129
130 if (!wait_for_echo(GPIO_PIN_SET, APP_ECHO_TIMEOUT_US)) {
131 return 0;
132 }
133 pulse_start = __HAL_TIM_GET_COUNTER(&echo_timer);
134 if (!wait_for_echo(GPIO_PIN_RESET, APP_ECHO_TIMEOUT_US)) {
135 return 0;
136 }
137 pulse_width = timer_elapsed(pulse_start);
138 if (pulse_width == 0U || pulse_width > APP_ECHO_TIMEOUT_US) {
139 return 0;
140 }
141
142 /* HC-SR04 pulse width is round-trip time: cm ~= us / 58. */
143 *distance_cm = ((uint32_t)pulse_width + 29U) / 58U;
144 return 1;
145}

147void app_start(void)
148{
149 uint32_t distance_cm;
150 uint32_t next_update = HAL_GetTick();
151 uint32_t caution_toggle = HAL_GetTick();
152 char message[64];
153 uint8_t caution_led = 0;
154
155 peripherals_init();

157
158 for (;;) {
159 uint32_t now = HAL_GetTick();
160 if ((int32_t)(now - next_update) >= 0) {
161 next_update = now + APP_DISTANCE_UPDATE_MS;
162 if (!measure_distance_cm(&distance_cm)) {
163 led_set(0);
164 print_line("Distance: timeout\r\n");
165 } else if (distance_cm < APP_DISTANCE_NEAR_CM) {
166 led_set(1);
167 (void)snprintf(message, sizeof(message),
168 "Distance: %lu cm\r\nSTOP: too close\r\n",
169 (unsigned long)distance_cm);
170 print_line(message);
171 } else if (distance_cm <= APP_DISTANCE_CAUTION_CM) {
172 (void)snprintf(message, sizeof(message),
173 "Distance: %lu cm\r\nCaution: object approaching\r\n",
174 (unsigned long)distance_cm);
175 print_line(message);
176 if ((int32_t)(now - caution_toggle) >= 0) {
177 caution_toggle = now + 500U;
178 caution_led = (uint8_t)!caution_led;
179 led_set(caution_led);
180 }
181 } else {
182 led_set(0);
183 (void)snprintf(message, sizeof(message),
184 "Distance: %lu cm\r\nClear\r\n",
185 (unsigned long)distance_cm);
186 print_line(message);
187 }
188 }
189 }
